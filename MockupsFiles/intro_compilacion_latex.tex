% COLUMNA 1: INSTRUCCIONES, OBJETIVO Y MARCO
\section*{\color{primarydark}\Large 1. Instrucciones y Objetivo}

\noindent\fbox{%
\parbox{0.96\linewidth}{%
\textbf{\color{primarydark}Instrucciones:}\\[2bp]
\small Como parte del proyecto de grado en Ingeniería de Sistemas Informáticos, se solicita diseñar los mockups que incluya tanto la versión web como la versión móvil (De acuerdo a cada proyecto).
}%
}

\vspace{6bp}

\noindent\fbox{%
\parbox{0.96\linewidth}{%
\textbf{\color{primarydark}El objetivo:}\\[2bp]
\small El objetivo es que los mockups reflejen claramente la funcionalidad principal del sistema, la experiencia de usuario, y la adaptación a diferentes dispositivos. Se debe mostrar cómo interactúan los usuarios con el sistema y cómo se garantizan los objetivos funcionales del mismo en ambas plataformas.
}%
}

\vspace{6bp}
\subsection*{\color{primary}Indicaciones Metodológicas}
\begin{itemize}[leftmargin=12bp, itemsep=2bp]
    \item \textbf{Diseña mockups para la interfaz web y móvil:} Láminas que unifican Web Desktop (1440$\times$900), Tablet (1024$\times$768) y Móvil PWA (390$\times$844) en una misma página de 2400$\times$1500 px.
    \item \textbf{Pantallas y flujos clave:} 10 pantallas críticas del SOC modeladas con interacciones reales y telemetría de kernel Linux.
    \item \textbf{Coherencia funcional:} La lógica de negocio, contratos de API REST y respuestas eBPF/XDP son idénticos en todas las plataformas.
    \item \textbf{Herramientas y técnicas adecuadas:} Modelado gráfico vectorial SVG, compilación automatizada y tipografía tabular sin distorsión.
    \item \textbf{Decisiones de diseño:} Enfoque riguroso en usabilidad, ergonomía y prevención de errores en salas de operaciones 24/7.
\end{itemize}

% COLUMNA 2: DECISIONES DE DISEÑO, USABILIDAD Y FUNCIONALIDAD
\section*{\color{primarydark}\Large 2. Usabilidad y Funcionalidad}

\subsection*{\color{primary}Erradicación de Anti-Patrones UI/UX de IA}
\begin{itemize}[leftmargin=12bp, itemsep=2bp]
    \item \textbf{Cero Emojis:} Eliminación absoluta de emojis informales en estados, tablas y botones. Se emplea iconografía técnica vectorial SVG pura y texto explícito.
    \item \textbf{Textos y Especificaciones Semi-Planos:} Contenedores con bordes uniformes de 1\,px (\emph{sin barras gruesas a la izquierda}). Eliminación del anidamiento excesivo (\emph{cardocalypse}).
    \item \textbf{Radios Concéntricos Calibrados:} Geometría armónica según $\text{radio}_{\text{interior}} = \max(0, \text{radio}_{\text{exterior}} - \text{padding})$.
    \item \textbf{Tipografía Tabular de Red:} Fuentes sans-serif limpias y tipografías monoespaciadas con cifras tabulares para IP, CIDR, hashes y pps.
\end{itemize}

\subsection*{\color{primary}Pilares de Seguridad Reforzada}
\begin{itemize}[leftmargin=12bp, itemsep=2bp]
    \item \textbf{Sesión Inmutable en Header:} Eliminación de cambio arbitrario de cuenta. Despliegue fijo de \texttt{Carlos Mendoza [admin\_soc]}, rol \texttt{SOC\_ADMIN}, estado TLS 1.3 y botón de \texttt{[Cerrar Sesión]}.
    \item \textbf{Rotación Obligatoria de Contraseña:} Detección de clave temporal por defecto (\texttt{admin}). Bloqueo de acceso hasta definir clave con medidor de entropía (94 bits) y checklist ISO 27001.
    \item \textbf{Autenticación Corporativa de 3 Factores:} Correo SSO institucional, casillas numéricas 2FA/TOTP sincronizadas con app móvil y rate limiting (3 intentos).
    \item \textbf{Sandbox ACL en Subred Aislada:} Subred virtualizada \texttt{10.200.0.0/24 (VLAN 999)} con inyector sintético, consola eBPF/iptables y promoción auditada.
\end{itemize}
